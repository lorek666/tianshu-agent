def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def mapping(root):
    return {str(p.relative_to(root)): sha(p) for p in sorted(root.rglob('*')) if p.is_file()}

def check_record(path, record):
    assert path.is_file(), str(path)
    assert path.stat().st_size == record['bytes'], str(path)
    assert sha(path) == record['sha256'], str(path)

def ast_without_constructor_wiring(text):
    tree = ast.parse(text)
    count = 0
    for node in ast.walk(tree):
        if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and node.func.id == 'AgentLoop':
            added = [kw for kw in node.keywords if kw.arg in {'memory', 'contents'}]
            for kw in added:
                assert isinstance(kw.value, ast.Attribute) and kw.value.attr == kw.arg
                expected = ast.parse('self.stack' if 'self.stack' in text else 'stack').body[0].value
                assert ast.dump(kw.value.value) == ast.dump(expected)
            assert {kw.arg for kw in added} == {'memory', 'contents'}
            count += len(added)
            node.keywords = [kw for kw in node.keywords if kw.arg not in {'memory', 'contents'}]
    return ast.dump(tree, include_attributes=False), count

def enums(x):
    if isinstance(x, dict):
        if 'enum' in x: yield set(x['enum'])
        for v in x.values(): yield from enums(v)
    elif isinstance(x, list):
        for v in x: yield from enums(v)
