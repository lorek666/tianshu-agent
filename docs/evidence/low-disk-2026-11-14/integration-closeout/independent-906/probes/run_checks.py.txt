expected = json.loads((OUT / 'inputs/source-hashes.json').read_text())

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def mapping(root):
    return {str(p.relative_to(root)): sha(p) for p in root.rglob('*') if p.is_file()}

def run(label, command, *, source=WORK, mutation=False, expected_exit=0, needle=None):
    (OUT / 'runs' / label).mkdir(parents=True, exist_ok=True)
    env = dict(os.environ)
    # No real model may be selected by this independent offline review.
    for key in ['TIANSHU_API_KEY', 'TIANSHU_BASE_URL', 'TIANSHU_MODEL_NAME', 'TIANSHU_EVAL_REAL_MODEL']:
        env.pop(key, None)
    env.update(PYTHONDONTWRITEBYTECODE='1', PYTHONPATH=f'{source / "src"}:{source / "tests"}',
               TIANSHU_AUTHORITY_ROOT=str(OUT / 'runs' / label / 'authority'),
               TIANSHU_SANDBOX_BACKEND='local_fs', REVIEW_LABEL=label,
               REVIEW_SOURCE=str(source), REVIEW_MUTATION='1' if mutation else '')
    meta = {'label': label, 'command': command, 'cwd': str(source),
            'environment_overrides': {k: env[k] for k in ['PYTHONDONTWRITEBYTECODE', 'PYTHONPATH',
                'TIANSHU_AUTHORITY_ROOT', 'TIANSHU_SANDBOX_BACKEND', 'REVIEW_LABEL', 'REVIEW_SOURCE', 'REVIEW_MUTATION']},
            'expected_exit': expected_exit, 'expected_output_contains': needle}
    (OUT / 'logs' / f'{label}.command.json').write_text(json.dumps(meta, indent=2) + '\n')
    start = time.monotonic()
    proc = subprocess.run(command, cwd=source, env=env, capture_output=True, text=True, timeout=180)
    (OUT / 'logs' / f'{label}.stdout.txt').write_text(proc.stdout)
    (OUT / 'logs' / f'{label}.stderr.txt').write_text(proc.stderr)
    combined = proc.stdout + proc.stderr
    meta.update(exit_code=proc.returncode, elapsed_seconds=round(time.monotonic()-start, 3),
                expected_behavior_observed=(proc.returncode == expected_exit and (needle is None or needle in combined)))
    RESULTS.append(meta)
    (OUT / 'logs/check-results.json').write_text(json.dumps(RESULTS, ensure_ascii=False, indent=2) + '\n')
    print(label, 'exit=', proc.returncode, 'expected=', meta['expected_behavior_observed'], flush=True)
    if not meta['expected_behavior_observed']:
        print(combined[-6000:], flush=True)
        raise AssertionError(f'{label}: wrong behavior; see raw stdout/stderr')
    return proc

    return [PYTHON, '-B', '-m', 'pytest', '-p', 'no:cacheprovider', '-o', 'addopts=', '-q', '-rs',
            '--basetemp', str(OUT / 'runs' / label / 'pytest'), *selectors]
