def env(root,free):
 e=os.environ.copy()

  if k.startswith(('TIANSHU_','OPENAI_','ANTHROPIC_')):e.pop(k)

def snapshot(root):
 db=root/'events.db'
 if not db.exists():return {'tables':{},'events':[],'versions':[],'manifests':[]}
 c=sqlite3.connect(db.as_uri()+'?mode=ro',uri=True);c.row_factory=sqlite3.Row
 try:
  names={r[0] for r in c.execute("SELECT name FROM sqlite_master WHERE type='table'")}
  tables={n:[dict(x) for x in c.execute('SELECT * FROM '+n)] for n in ['tasks','runs','jobs','leases','approvals','grants'] if n in names}
  events=[dict(x) for x in c.execute('SELECT seq,type,run_id,task_id,payload FROM events ORDER BY seq')]
  for e in events:e['payload']=json.loads(e['payload']) if e['payload'] is not None else None
  versions=[dict(x) for x in c.execute('SELECT * FROM event_versions ORDER BY seq')]
  return {'tables':tables,'events':events,'versions':versions,'manifests':[str(p.relative_to(root)) for p in root.rglob('manifests/*.json')]}
 finally:c.close()
def no_new_run(s):

def policy_snapshot(s,mode):

 e=starts[0];assert e['payload']['execution_policy']=={'format_version':1,'mode':mode}

  assert not s['manifests']

class Daemon:
 def __init__(self,free):
  self.root=case_root();(self.root/'workspace').mkdir();(self.root/'workspace/note').write_text('IPC original')
  self.err=(self.root/'daemon.stderr').open('w');self.p=subprocess.Popen([sys.executable,str(PROBES/'daemon_child.py')],cwd=WORK,env=env(self.root,free),stdout=subprocess.PIPE,stderr=self.err,text=True)
  ready,_,_=select.select([self.p.stdout],[],[],12)
  assert ready,'daemon readiness timed out'
  self.ready=self.p.stdout.readline();assert json.loads(self.ready)['ready'],self.ready
  self.c=ipc.Client(self.root,timeout=5);self.welcome=self.c.connect()
  self.task=self.c.call('task.create',{'goal':'independent real IPC','workspaces':[{'root_path':str(self.root/'workspace'),'mode':'writable'}]})['task']
  emit('ipc_ready',pid=self.p.pid,parent=os.getpid(),root=self.root,welcome=self.welcome)
 def close(self):
  try:self.c.call('shutdown',{'reason':'independent T3 complete'})
  finally:self.c.close()
  stdout,_=self.p.communicate(timeout=12);self.err.close();assert self.p.returncode==0
  (self.root/'daemon.stdout').write_text(self.ready+stdout)
 def start(self,policy=OMIT,extra=None):
  params={'task_id':self.task['task_id']}
  if policy is not OMIT:params['execution_policy']=policy
  if extra:params.update(extra)
  try:r=self.c.call('run.start',params);return {'ok':True,'result':r}
  except ipc.RemoteError as e:return {'ok':False,'code':e.code,'message':str(e)}
 def settled(self,rid):
  deadline=time.monotonic()+8
  while time.monotonic()<deadline:
   r=self.c.call('run.status',{'run_id':rid})['run']
   if r['state'] in ['succeeded','failed','cancelled','completed','fenced','waiting_approval','paused']:return r
  raise AssertionError('run did not settle')

@pytest.mark.parametrize('free',[MIN-1,MIN,MIN+1,0,True,-1,'raise'])
def test_real_ipc_default_threshold(free):
 d=Daemon(free)
 try:
  result=d.start()
  if type(free)is int and free>=MIN:
   assert result['ok'];assert result['result']['run']['state']=='created';d.settled(result['result']['run']['run_id'])

 finally:d.close()

 if result['ok']:policy_snapshot(s,'effects')
 else:no_new_run(s);assert not (d.root/'model.jsonl').exists()
