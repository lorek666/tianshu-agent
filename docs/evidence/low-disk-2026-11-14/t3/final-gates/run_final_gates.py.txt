BASE_ENV = {
    **os.environ,
    "PYTHONDONTWRITEBYTECODE": "1",
    "PYTEST_ADDOPTS": "-p no:cacheprovider",
}

results: list[dict] = []
for label, argv, overrides in COMMANDS:
    env = {**BASE_ENV, **overrides}
    completed = subprocess.run(
        argv,
        cwd=ROOT,
        env=env,
        text=True,
        capture_output=True,
        check=False,
    )
    (OUT / f"{label}.stdout.txt").write_text(completed.stdout, encoding="utf-8")
    (OUT / f"{label}.stderr.txt").write_text(completed.stderr, encoding="utf-8")
    (OUT / f"{label}.exit-code.txt").write_text(
        f"{completed.returncode}\n", encoding="utf-8"
    )
    metadata = {
        "label": label,
        "cwd": str(ROOT),
        "argv": argv,
        "exit_code": completed.returncode,
        "env": {
            "PYTHONDONTWRITEBYTECODE": env["PYTHONDONTWRITEBYTECODE"],
            "PYTEST_ADDOPTS": env["PYTEST_ADDOPTS"],
        },
        "stdout": f"{label}.stdout.txt",
        "stderr": f"{label}.stderr.txt",
    }
    (OUT / f"{label}.meta.json").write_text(
        json.dumps(metadata, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )
    results.append(metadata)
    tail = (completed.stdout + completed.stderr).strip().splitlines()
    print(f"{label}: exit={completed.returncode} {' | '.join(tail[-2:])}")

(OUT / "results.json").write_text(
    json.dumps(
        {
            "head": subprocess.check_output(
                ["git", "rev-parse", "HEAD"], cwd=ROOT, text=True
            ).strip(),
            "commands": results,
        },
        ensure_ascii=False,
        indent=2,
    )
    + "\n",
    encoding="utf-8",
)
if any(item["exit_code"] != 0 for item in results):
    raise SystemExit(1)
