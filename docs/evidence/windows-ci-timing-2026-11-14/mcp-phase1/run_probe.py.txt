"""Bounded isolated invocation; preserves raw output and exact command metadata."""
import hashlib
import json
import os
from pathlib import Path
import platform
import subprocess
import sys
import time

root = Path(__file__).resolve().parent
mode = sys.argv[1]
assert mode in {"observe", "gate-approval", "gate-unknown"}
assert not (root / f"{mode}.run.json").exists(), "preserve the original probe"
python = Path("/Users/ronfee/.codex/worktrees/g3-data-admission/天枢Agent/.venv/bin/python3")
command = [
    str(python), "-B", "-m", "pytest", "-p", "boundary_plugin", "-p", "no:cacheprovider",
    "tests/test_mcp_transport_timeout.py::test_stalled_send_is_unknown_and_replay_does_not_retry",
    "-o", "addopts=", "-q", "--tb=short", "--basetemp", str(root / f"{mode}-pytest"),
]
env = os.environ.copy()
env.update({
    "PYTHONPATH": str(root),
    "PYTHONDONTWRITEBYTECODE": "1",
    "PYTEST_DISABLE_PLUGIN_AUTOLOAD": "1",
    "MCP_BOUNDARY_MODE": mode,
    "MCP_BOUNDARY_OUTPUT": str(root / f"{mode}.json"),
})
start = time.monotonic_ns()
metadata = {"command": command, "cwd": str(root / "source"), "mode": mode,
            "platform": platform.platform(), "plugin_sha256": hashlib.sha256(
                (root / "boundary_plugin.py").read_bytes()).hexdigest()}
try:
    result = subprocess.run(command, cwd=root / "source", env=env, timeout=60,
                            stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    output = result.stdout
    metadata["returncode"] = result.returncode
except subprocess.TimeoutExpired as error:
    output = error.stdout or b""
    metadata.update({"returncode": None, "timeout_seconds": 60})
metadata["elapsed_ns"] = time.monotonic_ns() - start
(root / f"{mode}.log").write_bytes(output)
metadata["log_sha256"] = hashlib.sha256(output).hexdigest()
(root / f"{mode}.run.json").write_text(json.dumps(metadata, ensure_ascii=False, indent=2) + "\n")
print(json.dumps(metadata, ensure_ascii=False, indent=2))
print(output.decode(errors="replace"))
raise SystemExit(metadata["returncode"] if metadata["returncode"] is not None else 124)
