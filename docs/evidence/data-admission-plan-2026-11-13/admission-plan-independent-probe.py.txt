def quote(name):
    return '"' + name.replace('"', '""') + '"'


def typed(v):
    if v is None:
        return ['null']
    if isinstance(v, int):
        return ['integer', str(v)]
    if isinstance(v, float):
        return ['float64', struct.pack('>d', v).hex()]
    if isinstance(v, str):
        return ['text', v]
    if isinstance(v, bytes):
        return ['blob', v.hex()]
    raise TypeError(type(v).__name__)


def digest(file):
    conn = sqlite3.connect(file)
    try:
        schema = conn.execute(
            'SELECT type,name,tbl_name,sql FROM sqlite_schema '
            'WHERE name NOT IN (?,?) ORDER BY type,name,tbl_name,sql',
            ('migration_log', 'idx_migration_log_applied'),
        ).fetchall()
        tables = [r[1] for r in schema if r[0] == 'table']
        contents = []
        for name in tables:
            values = [
                [typed(v) for v in row]
                for row in conn.execute('SELECT rowid,* FROM ' + quote(name))
            ]
            values.sort(key=lambda row: json.dumps(row, ensure_ascii=False, separators=(',', ':')))
            contents.append([name, values])
        data = json.dumps([schema, contents], ensure_ascii=False, separators=(',', ':')).encode()
        return hashlib.sha256(data).hexdigest()
    finally:
        conn.close()


def mechanism(file):
    conn = sqlite3.connect(file)
    try:
        version = conn.execute('PRAGMA user_version').fetchone()[0]
        schema = conn.execute(
            'SELECT type,name,tbl_name,sql FROM sqlite_schema '
            'WHERE name IN (?,?) ORDER BY type,name,tbl_name,sql',
            ('migration_log', 'idx_migration_log_applied'),
        ).fetchall()
        rows = conn.execute('SELECT step,applied_at,note FROM migration_log ORDER BY step').fetchall()
        return [version, schema, [[typed(v) for v in r] for r in rows]]
    finally:
        conn.close()
