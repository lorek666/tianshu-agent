def ident(value):
    return '"'+value.replace('"','""')+'"'

def atom(value):
    if value is None: return ['null']
    if type(value) is int: return ['int',str(value)]
    if type(value) is float: return ['float64',struct.pack('>d',value).hex()]
    if type(value) is str: return ['text',value]
    if type(value) is bytes: return ['blob',value.hex()]
    raise TypeError(type(value).__name__)

def encoded(value):
    return json.dumps(value,ensure_ascii=False,separators=(',',':')).encode()

def logical(conn):
    # Use one read transaction. Do not turn on immutable or checkpoint live WAL.
    conn.execute('BEGIN')
    try:
        integrity=[r[0] for r in conn.execute('PRAGMA integrity_check')]
        assert integrity==['ok'],integrity
        excluded={'migration_log','idx_migration_log_applied'}
        objects=[list(row) for row in conn.execute('SELECT type,name,tbl_name,sql FROM sqlite_schema ORDER BY type,name') if row[1] not in excluded]
        flags={r[1]:r[4] for r in conn.execute('PRAGMA main.table_list')}
        tables=[]
        for _,name,_,_ in objects:
            if not any(obj[0]=='table' and obj[1]==name for obj in objects) or name in [t['name'] for t in tables]: continue
            columns=[list(r) for r in conn.execute('PRAGMA table_xinfo('+ident(name)+')')]
            selected=[str(c[1]) for c in columns if c[6]!=1]
            names={str(c[1]) for c in columns}
            rowid=None
            if flags.get(name)==0:
                rowid=next((candidate for candidate in ('rowid','_rowid_','oid') if candidate not in names),None)
                if rowid is None: raise ValueError('All rowid aliases shadowed')
            expressions=([rowid] if rowid else [])+[ident(c) for c in selected]
            rows=sorted(encoded([atom(v) for v in row]) for row in conn.execute('SELECT '+','.join(expressions)+' FROM '+ident(name)))
            tables.append({'name':name,'columns':columns,'rowid':bool(rowid),'rows':[r.decode() for r in rows]})
        payload={'prototype_algorithm':'all-tables-typed-v1','objects':objects,'tables':tables}
        return hashlib.sha256(encoded(payload)).hexdigest(),[t['name'] for t in tables]
    finally:
        conn.rollback()

def materials(root):
    entries=[]
    for top in ('artifacts','blobs','memory_blobs'):
        for path in sorted((root/top).rglob('*')):
            if path.is_symlink(): raise ValueError('symlink')
            if path.is_file():
                raw=path.read_bytes()
                entries.append([str(path.relative_to(root)),len(raw),hashlib.sha256(raw).hexdigest()])
    return hashlib.sha256(encoded(entries)).hexdigest(),entries
