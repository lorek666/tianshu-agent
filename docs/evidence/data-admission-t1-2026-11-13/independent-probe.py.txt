def record(name, **detail):

def db_at(label, name='events.db', legacy=False):
    base = RUN / label
    base.mkdir(parents=True, exist_ok=True)
    db = base / name
    with sqlite3.connect(db) as con:
        con.executescript((FIXTURES / ('legacy-v2.sql' if legacy else 'v0.sql')).read_text(encoding='utf-8'))
    return db

def sql(db, statement, params=()):
    with sqlite3.connect(db) as con:
        con.execute(statement, params)

def v2_state(db, origin=0):
    return {
        'format': 'tianshu-migration', 'format_version': 2,
        'db_path': str(db.resolve()), 'from_version': origin, 'target_version': 2,
        'status': 'completed', 'applied_steps': list(range(origin + 1, 3)), 'current_step': None,
        'plan_sha256': 'a'*64,
        'backup': {'path': str((db.parent / '备份').resolve()), 'manifest_sha256': 'b'*64},
        'baseline': {'sqlite_algorithm': 'tianshu-sqlite-logical-v1', 'sqlite_sha256': 'c'*64,
                     'material_algorithm': 'tianshu-material-inventory-v1', 'material_sha256': 'd'*64},
        'started_at_ms': 1, 'updated_at_ms': 2,
    }

def sidecar(db, state=None):
    p = db.with_name(db.name + '.migration.json')
    p.write_text(json.dumps(state or v2_state(db), ensure_ascii=False), encoding='utf-8')
    return p

def oldstate(db):
    p = db.parent / 'migration.json'
    p.write_bytes((FIXTURES / 'legacy-completed.json').read_bytes())
    return p

def unchanged_state(db):
    # A logical WAL-aware snapshot; journal/SHM physical maintenance is not a product mutation.
    with sqlite3.connect(db.as_uri() + '?mode=ro', uri=True) as con:
        logical = tuple(con.iterdump())
        version = con.execute('PRAGMA user_version').fetchone()[0]
        journal = con.execute('PRAGMA journal_mode').fetchone()[0]
    files = {}
    for p in db.parent.rglob('*'):
        if p.is_symlink():
            files[str(p.relative_to(db.parent))] = ['symlink', os.readlink(p)]
        elif p.is_file() and p != db and not p.name.endswith(('-wal', '-shm', '-journal')):
            files[str(p.relative_to(db.parent))] = ['file', hashlib.sha256(p.read_bytes()).hexdigest()]
    return logical, version, journal, files

def refused(db, expected=None):
    before = unchanged_state(db)
    try:
        EventLog(db, blob_dir=db.parent/'new-payload')
    except TianshuError as exc:
        code = getattr(exc, 'code', None)
        if expected is not None:
            assert code == expected, (expected, code)
    else:
        raise AssertionError('unexpected admission: ' + str(db))
    assert unchanged_state(db) == before
    assert not (db.parent/'new-payload').exists()
    return code
